7	FATAL	De tool zelf is gecrasht of mist dependencies.



class StatusRenderer(BaseRenderer):
    VERSION = "1.1.1"
    
    # Jouw specifieke exit-code tabel
    EXIT_CODES = {
        "OK": 0,
        "WARNING": 1,
        "EXPIRED": 2,
        "INCOMPLETE": 3,
        "INVALID": 4,
        "REVOKED": 5,
        "INPUT_ERR": 6,
        "FATAL": 7
    }

    def render(self, analysis_results: Any, **kwargs) -> str:
        try:
            report_groups = []
            max_exit_code = 0
            scan_now = datetime.now(timezone.utc)

            for group_data in analysis_results:
                g_name = group_data.get("group_name") or group_data.get("groupName") or "unknown"
                nodes = group_data.get("tree", [])
                
                # We flatten de boom om elke node te kunnen beoordelen
                flat_list = self._flatten_tree(nodes, group_name=g_name)
                all_certs = [item["cert"] for item in flat_list]
                
                chain_report = []
                group_max_code = 0

                for item in flat_list:
                    cert = item["cert"]
                    status_info = self._analyze_node_status(cert, all_certs)
                    
                    # Update de hoogste exit code (voor het hele rapport)
                    if status_info["exit_code"] > max_exit_code:
                        max_exit_code = status_info["exit_code"]
                    if status_info["exit_code"] > group_max_code:
                        group_max_code = status_info["exit_code"]

                    # Bouw de node-output (Conform jouw JSON voorbeeld)
                    chain_report.append({
                        "commonName": self._get_val(cert, "common_name", "commonName"),
                        "serialNumber": self._get_val(cert, "serial_number", "serialNumber") or "UNKNOWN",
                        "isSystemCert": self._get_val(cert, "is_system_cert", "isSystemCert") or False,
                        "signatureValid": self._get_val(cert, "signature_valid", "signatureValid"),
                        "expiryDate": self._format_date(self._get_val(cert, "expiry_date", "expiryDate")),
                        "trustStatus": status_info["label"]
                    })

                # Groeps-samenvatting
                report_groups.append({
                    "groupName": g_name,
                    "groupStatus": self._get_label_by_code(group_max_code),
                    "summary": {
                        "totalCertificates": len(chain_report),
                        "isChainComplete": group_max_code != self.EXIT_CODES["INCOMPLETE"],
                        "isTrusted": group_max_code <= self.EXIT_CODES["WARNING"]
                    },
                    "chain": chain_report
                })

            # Het uiteindelijke rapport-jasje
            final_report = {
                "metadata": {
                    "version": self.VERSION,
                    "scanDate": scan_now.isoformat().replace("+00:00", "Z"),
                    "exitCode": max_exit_code
                },
                "groups": report_groups
            }

            return json.dumps(final_report, indent=2, cls=DateTimeEncoder)

        except Exception as e:
            # Code 7: FATAL
            return json.dumps({
                "metadata": {"version": self.VERSION, "exitCode": 7},
                "error": "Status rendering fatal error",
                "details": str(e)
            }, indent=2)

    def _analyze_node_status(self, cert: Any, all_certs: List) -> Dict:
        """Bepaalt de status en exit-code per certificaat."""
        v_err = self._get_val(cert, "validation_error") or ""
        sig_valid = self._get_val(cert, "signature_valid")
        expiry = self._get_val(cert, "expiry_date", "expiryDate")
        is_soon = self._get_val(cert, "is_expiring_soon", "isExpiringSoon")
        
        # 4. INVALID (Handtekening of logica)
        if sig_valid is False or "PARENT_NOT_A_CA" in v_err:
            return {"exit_code": 4, "label": "INVALID"}
            
        # 3. INCOMPLETE (Missing links / Orphan)
        if self._get_val(cert, "common_name") == ORPHAN_NODE_ID:
            return {"exit_code": 3, "label": "INCOMPLETE"}

        # 2. EXPIRED
        if expiry and isinstance(expiry, datetime):
            if expiry < datetime.now(timezone.utc):
                return {"exit_code": 2, "label": "EXPIRED"}

        # 1. WARNING (Expiring soon)
        if is_soon:
            return {"exit_code": 1, "label": "WARNING"}

        # 0. OK
        return {"exit_code": 0, "label": "TRUSTED"}

    def _get_val(self, obj, attr, alias=None):
        if isinstance(obj, dict):
            return obj.get(alias or attr, obj.get(attr))
        return getattr(obj, attr, None)

    def _format_date(self, d):
        if hasattr(d, "isoformat"):
            return d.isoformat().replace("+00:00", "Z")
        return str(d).replace("+00:00", "Z")

    def _get_label_by_code(self, code):
        for label, c in self.EXIT_CODES.items():
            if c == code: return label
        return "UNKNOWN"
